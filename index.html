// دالة استخراج كائن request المباشر بدقة متناهية function extractRequestObject(result) { if (!result) return {}; // 1. إذا كانت الاستجابة مغلّفة بـ data let body = result.data !== undefined ? result.data : result; // 2. إذا كانت الاستجابة مصفوفة Array (كما يظهر باللوغ لديك [ {...} ]) if (Array.isArray(body) && body.length > 0) { body = body[0]; } // 3. الوصول إلى كائن request بداخل الاستجابة if (body && body.request) { return body.request; } return body || {}; } function renderVehicleCard(apiResult) { const resultContainer = document.getElementById('resultContainer'); // استخراج كائن request المباشر const req = extractRequestObject(apiResult); console.log('بيانات المركبة المستخرجة بنجاح:', req); // دالة لتنظيف النص وعرض القيمة إن وجدت function getVal(val) { if (val === null || val === undefined) return ''; const str = String(val).trim(); return str; } // ربط الحقول بالمتغيرات المباشرة من request const maker = getVal(req.vehicleMaker); const model = getVal(req.vehicleModel); const year = getVal(req.modelYear); const seq = getVal(req.sequenceNumber); const vin = getVal(req.vin); const customId = getVal(req.customId); const expiryDate = getVal(req.expiryDate); const requestId = getVal(req.id); const source = getVal(req.source); // تجهيز اللوحة let plateStr = getVal(req.plateDesc); if (!plateStr || plateStr === '0000') { const l1 = getVal(req.letterOne); const l2 = getVal(req.letterTwo); const l3 = getVal(req.letterThree); const pNum = getVal(req.plateNumber); if (l1 || l2 || l3 || (pNum && pNum !== '0')) { plateStr = `${l1} ${l2} ${l3} ${pNum}`.trim(); } else { plateStr = ''; } } const title = (maker || model) ? `${maker} ${model}`.trim() : 'بيانات المركبة المستعلم عنها'; resultContainer.innerHTML = `

${title}

${year ? `سنة الصنع: ${year}` : ''}
الشركة المصنعة ${maker}
موديل المركبة ${model}
الرقم التسلسلي ${seq}
رقم اللوحة ${plateStr}
رقم الهيكل (VIN) ${vin}
الرقم الجمركي ${customId}
تاريخ الانتهاء ${expiryDate}
رقم الطلب ${requestId}
المصدر ${source}
`; resultContainer.style.display = 'block'; }
